\begin{frame}{Несобственные интегралы}

\begin{block}{Первое свойство:}
\begin{equation}
\int_{a}^{\omega} f(x) \, dx = \int_{\alpha}^{\gamma} (f \circ \varphi)(t) \varphi'(t) \, dt  
\end{equation}
\end{block}

\begin{block}{Второе свойство:}
\begin{equation}
\int_{a}^{\omega} (f \cdot g')(x) \, dx = (f \cdot g)(x)\big|_{a}^{\omega} - \int_{a}^{\omega} (f' \cdot g)(x) \, dx
\end{equation}
\end{block}
\end{frame}


\begin{frame}{Важные табличные интегралы}
Интегралы тригонометрических функций:
\begin{enumerate} 
    \item \textbf{arcsin:} 
   $$\int \frac{dx}{\sqrt{a^2 - x^2}} = \arcsin \frac{x}{a} + C, \quad |x| < |a|$$
    \item \textbf{arctg:} 
   $$\int \frac{dx}{a^2 + x^2} = \frac{1}{a} \operatorname{arctg} \frac{x}{a} + C$$
    \item \textbf{ctg:} 
    $$\int \frac{dx}{\sin^2 x} = -\operatorname{ctg} x + C$$
    \item \textbf{tg:}
    $$\int \frac{dx}{\cos^2 x} = \operatorname{tg} x + C$$
\end{enumerate} 
\end{frame}
